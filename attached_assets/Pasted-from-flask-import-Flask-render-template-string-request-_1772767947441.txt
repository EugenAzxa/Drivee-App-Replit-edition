from datetime import datetime

app = Flask(__name__)

# In-memory database for demonstration purposes. 
# For a real app, use a database like SQLite or PostgreSQL.
users_data = []

# HTML Template for the dashboard

        .pay-btn { display: inline-block; margin-top: 10px; padding: 10px 15px; background-color: #28a745; color: white; text-decoration: none; border-radius: 4px; }

                <label>License Plate:</label>

@app.route('/')
def index():
    # Render the dashboard with the current ticket data
    return render_template_string(HTML_TEMPLATE, users_data=users_data)

@app.route('/add', methods=['POST'])
def add_ticket():
    # Extract data from the form
    name = request.form.get('name')
    plate = request.form.get('plate')
    ticket_num = request.form.get('ticket_num')
    due_date = request.form.get('due_date')

    # Save to our "database"
    users_data.append({
        'name': name,
        'plate': plate.upper(),
        'ticket_num': ticket_num,
        'due_date': due_date
    })

    # Redirect back to the homepage to see the updated list
    return redirect(url_for('index'))

if __name__ == '__main__':
    # Run the Flask server. Replit uses port 8080 by default.
    app.run(host='0.0.0.0', port=8080)
