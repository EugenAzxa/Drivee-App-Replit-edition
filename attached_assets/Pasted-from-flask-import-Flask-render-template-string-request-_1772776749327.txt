    <div id="gpsAlertBanner" class="gps-alert">

    <script>
        // 1. Initialize Map
        // Default to Downtown Toronto (Richmond & Spadina area)
        var map = L.map('map').setView([43.648, -79.396], 16);
        L.tileLayer('https://{s}.basemaps.cartocdn.com/dark_all/{z}/{x}/{y}{r}.png').addTo(map);

        // 2. Add "Hazard" Data (Simulating Open Data)
        // Bike Lane (Red Line) - Representing Richmond St W Cycle Track
        var bikeLaneCoords = [
            [43.6475, -79.3980], 
            [43.6490, -79.3920]
        ];
        var bikeLane = L.polyline(bikeLaneCoords, {color: '#da3633', weight: 6, opacity: 0.8}).addTo(map);
        bikeLane.bindPopup("Protected Bike Lane - $200 Fine Zone");

        // Fire Hydrant (Yellow Circle)
        var hydrantPos = [43.6485, -79.3950];
        var hydrant = L.circleMarker(hydrantPos, {color: '#d29922', radius: 8, fillOpacity: 1}).addTo(map);
        hydrant.bindPopup("Fire Hydrant - Must be 3m away ($100 Fine)");

        // 3. User Tracking Variables
        var userMarker = null;
        var watchId = null;

        // 4. GPS Guardian Logic
        function startGPSGuardian() {
            var btn = document.getElementById('gpsBtn');
            btn.innerHTML = "<i class='fa-solid fa-spinner fa-spin'></i> Locating...";
            btn.style.background = "#238636"; // Turn green

            if ("geolocation" in navigator) {
                // Start tracking the user
                watchId = navigator.geolocation.watchPosition(updatePosition, handleError, {
                    enableHighAccuracy: true,
                    maximumAge: 0
                });
            } else {
                alert("Geolocation is not supported by your browser.");
            }
        }

        function updatePosition(position) {
            document.getElementById('gpsBtn').innerHTML = "<i class='fa-solid fa-satellite-dish'></i> Guardian Active";
            
            var lat = position.coords.latitude;
            var lng = position.coords.longitude;
            var userLatLng = L.latLng(lat, lng);

            // Update User Marker (Blue Dot)
            if (!userMarker) {
                userMarker = L.circleMarker(userLatLng, {color: '#58a6ff', radius: 8, fillOpacity: 1}).addTo(map);
                map.setView(userLatLng, 17);
            } else {
                userMarker.setLatLng(userLatLng);
            }

            // --- PROXIMITY CHECKING ---
            var alertBanner = document.getElementById('gpsAlertBanner');
            var alertText = document.getElementById('gpsAlertText');
            var isHazard = false;

            // Check distance to Hydrant (in meters)
            var distToHydrant = userLatLng.distanceTo(L.latLng(hydrantPos[0], hydrantPos[1]));
            
            // Check distance to middle of Bike Lane (Simplified for Demo)
            var distToBikeLane = userLatLng.distanceTo(L.latLng(43.6482, -79.3950)); 

            if (distToHydrant < 20) {
                alertText.innerHTML = "PULL FORWARD! <br> You are too close to a Fire Hydrant ($100 Fine).";
                alertBanner.style.display = "block";
                isHazard = true;
            } else if (distToBikeLane < 25) {
                alertText.innerHTML = "DO NOT STOP! <br> You are in a Protected Bike Lane ($200 Fine).";
                alertBanner.style.display = "block";
                isHazard = true;
            } 
            
            if (!isHazard) {
                alertBanner.style.display = "none";
            }
        }

        function handleError(error) {
            alert("Error accessing location. Please ensure Location Services are enabled for your browser.");
            document.getElementById('gpsBtn').innerHTML = "Start Live GPS Guardian";
            document.getElementById('gpsBtn').style.background = "linear-gradient(45deg, #1f6feb, #58a6ff)";
        }
    </script>
</body>
</html>
