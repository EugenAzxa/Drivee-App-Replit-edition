@app.route('/')
def index():
    # Convert the Python list to a JSON string so JavaScript can read it to draw the map pins
    reports_json = json.dumps(reports_data)
    return render_template_string(HTML_TEMPLATE, reports_json=reports_json)

@app.route('/report_311', methods=['POST'])
def handle_311_report():
    # Grab the data sent from the JavaScript GPS tracker
    issue_type = request.form.get('issue_type')
    lat = request.form.get('lat')
    lng = request.form.get('lng')
    
    # Save it to our database
    if issue_type and lat and lng:
        reports_data.append({
            "type": issue_type,
            "lat": float(lat),
            "lng": float(lng),
            "status": "Logged in Community App"
        })
        
    # Refresh the page to show the new pin
    return redirect(url_for('index'))
