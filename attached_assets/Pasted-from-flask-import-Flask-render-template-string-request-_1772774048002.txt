        // --- AI SCANNING LOGIC ---
        async function performOCR(event) {
            const file = event.target.files[0];
            if (!file) return;

            document.getElementById('loading').style.display = 'block';
            
            // Start AI process
            const result = await Tesseract.recognize(file, 'eng');
            const text = result.data.text;
            console.log("Extracted Text:", text);

            // Simple Pattern Matching for Toronto Plates & Dates
            // Look for 4 letters followed by 3 numbers (common Ontario format)
            const plateMatch = text.match(/[A-Z]{4}\s?\d{3}/);
            if(plateMatch) document.getElementById('plateInput').value = plateMatch[0];

            // Look for Date formats YYYY/MM/DD or MM/DD/YYYY
            const dateMatch = text.match(/\d{4}[\/\-]\d{2}[\/\-]\d{2}/);
            if(dateMatch) document.getElementById('dateInput').value = dateMatch[0].replace(/\//g, '-');

def add_ticket():

    return redirect(url_for('index'))
